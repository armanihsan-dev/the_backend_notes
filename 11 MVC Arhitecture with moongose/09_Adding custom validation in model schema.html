Mongoose Custom Validation: Advanced Techniques

1. BASIC CUSTOM VALIDATION

import mongoose from 'mongoose';
// ============================================
// 1. BASIC CUSTOM VALIDATION
// ============================================
const userSchema = new mongoose.Schema({
  
  // Method 1: Using validate object with validator function
  username: {
    type: String,
    required: true,
    validate: {
      validator: function(value) {
        // Must start with letter, only alphanumeric and underscore
        return /^[a-zA-Z][a-zA-Z0-9_]*$/.test(value);
      },
      message: 'Username must start with a letter and contain only letters, numbers, and underscores'
    }
  },
  
  // Method 2: Shorthand - function that returns boolean
  age: {
    type: Number,
    validate: {
      validator: function(value) {
        return value >= 18 && value <= 100;
      },
      message: 'Age must be between 18 and 100'
    }
  },
  
  // Method 3: Multiple validators for same field
  email: {
    type: String,
    required: true,
    validate: [
      {
        validator: function(value) {
          return /^\S+@\S+\.\S+$/.test(value);
        },
        message: 'Please enter a valid email address'
      },
      {
        validator: async function(value) {
          // Async validation - check if email already exists
          const user = await mongoose.models.User.findOne({ email: value });
          return !user;
        },
        message: 'Email already exists'
      }
    ]
  }
});

2. CUSTOM VALIDATION WITH PROPS ACCESS

// ============================================
// 2. CUSTOM VALIDATION WITH PROPS ACCESS
// ============================================
const productSchema = new mongoose.Schema({
  
  name: {
    type: String,
    required: true
  },
  
  price: {
    type: Number,
    required: true
  },
  
  discount: {
    type: Number,
    default: 0,
    validate: {
      validator: function(value) {
        // 'this' refers to the document
        // Discount cannot be greater than price
        return value <= this.price;
      },
      message: 'Discount cannot be greater than the price'
    }
  },
  
  // Validate based on another field
  salePrice: {
    type: Number,
    validate: {
      validator: function(value) {
        // Sale price must be less than regular price
        return value < this.price;
      },
      message: props => `Sale price (${props.value}) must be less than regular price`
    }
  }
});

3. ASYNC CUSTOM VALIDATION

// ============================================
// 3. ASYNC CUSTOM VALIDATION
// ============================================
const articleSchema = new mongoose.Schema({
  
  title: {
    type: String,
    required: true
  },
  
  slug: {
    type: String,
    required: true,
    validate: {
      validator: async function(value) {
        // Check if slug is unique (excluding current document on update)
        const article = await mongoose.models.Article.findOne({ 
          slug: value,
          _id: { $ne: this._id } // Exclude current document
        });
        return !article; // Return true if no duplicate found
      },
      message: 'Slug must be unique'
    }
  },
  
  authorId: {
    type: mongoose.Schema.Types.ObjectId,
    required: true,
    validate: {
      validator: async function(value) {
        // Check if author exists in database
        const User = mongoose.model('User');
        const author = await User.findById(value);
        return !!author;
      },
      message: 'Author does not exist'
    }
  }
});

4. CONDITIONAL VALIDATION

// ============================================
// 4. CONDITIONAL VALIDATION
// ============================================
const orderSchema = new mongoose.Schema({
  
  orderType: {
    type: String,
    enum: ['delivery', 'pickup'],
    required: true
  },
  
  // Only required if orderType is 'delivery'
  deliveryAddress: {
    type: String,
    required: function() {
      return this.orderType === 'delivery';
    },
    validate: {
      validator: function(value) {
        // Only validate if orderType is delivery
        if (this.orderType !== 'delivery') return true;
        
        // Address must be at least 10 characters
        return value && value.length >= 10;
      },
      message: 'Delivery address must be at least 10 characters long'
    }
  },
  
  // Only required if orderType is 'pickup'
  pickupTime: {
    type: Date,
    required: function() {
      return this.orderType === 'pickup';
    }
  }
});

5. ARRAY VALIDATION

// ============================================
// 5. ARRAY VALIDATION
// ============================================
const courseSchema = new mongoose.Schema({
  
  title: String,
  
  // Validate array length
  tags: {
    type: [String],
    validate: {
      validator: function(value) {
        return value.length >= 1 && value.length <= 5;
      },
      message: 'Must have between 1 and 5 tags'
    }
  },
  
  // Validate array elements
  ratings: {
    type: [Number],
    validate: {
      validator: function(array) {
        // Each rating must be between 1 and 5
        return array.every(rating => rating >= 1 && rating <= 5);
      },
      message: 'All ratings must be between 1 and 5'
    }
  },
  
  // Validate unique array values
  participants: {
    type: [mongoose.Schema.Types.ObjectId],
    validate: {
      validator: function(array) {
        // Check for duplicates
        return array.length === new Set(array.map(id => id.toString())).size;
      },
      message: 'Participants must be unique'
    }
  }
});

6. NESTED OBJECT VALIDATION

// ============================================
// 6. NESTED OBJECT VALIDATION
// ============================================
const profileSchema = new mongoose.Schema({
  
  name: String,
  
  socialMedia: {
    twitter: {
      type: String,
      validate: {
        validator: function(value) {
          if (!value) return true; // Optional field
          return /^@[a-zA-Z0-9_]{1,15}$/.test(value);
        },
        message: 'Twitter handle must start with @ and be 1-15 characters'
      }
    },
    
    linkedin: {
      type: String,
      validate: {
        validator: function(value) {
          if (!value) return true;
          return value.includes('linkedin.com/');
        },
        message: 'Invalid LinkedIn URL'
      }
    }
  },
  
  // Validate entire nested object
  address: {
    type: {
      street: String,
      city: String,
      zipCode: String
    },
    validate: {
      validator: function(value) {
        // If address exists, all fields must be provided
        if (!value) return true;
        return value.street && value.city && value.zipCode;
      },
      message: 'Address must include street, city, and zip code'
    }
  }
});

7. CUSTOM VALIDATION WITH CUSTOM ERROR

// ============================================
// 7. CUSTOM VALIDATION WITH CUSTOM ERROR
// ============================================
const passwordSchema = new mongoose.Schema({
  
  username: String,
  
  password: {
    type: String,
    required: true,
    validate: {
      validator: function(value) {
        // Complex password validation
        const hasUpperCase = /[A-Z]/.test(value);
        const hasLowerCase = /[a-z]/.test(value);
        const hasNumber = /[0-9]/.test(value);
        const hasSpecialChar = /[!@#$%^&*]/.test(value);
        const isLongEnough = value.length >= 8;
        
        return hasUpperCase && hasLowerCase && hasNumber && hasSpecialChar && isLongEnough;
      },
      message: 'Password must contain at least 8 characters, including uppercase, lowercase, number, and special character (!@#$%^&*)'
    }
  },
  
  confirmPassword: {
    type: String,
    required: true,
    validate: {
      validator: function(value) {
        // Compare with password field
        return value === this.password;
      },
      message: 'Passwords do not match'
    }
  }
});

8. SCHEMA-LEVEL VALIDATION (Multiple Fields)

// ============================================
// 8. SCHEMA-LEVEL VALIDATION (Multiple Fields)
// ============================================
const eventSchema = new mongoose.Schema({
  
  name: {
    type: String,
    required: true
  },
  
  startDate: {
    type: Date,
    required: true
  },
  
  endDate: {
    type: Date,
    required: true
  },
  
  maxParticipants: {
    type: Number,
    required: true
  },
  
  registeredParticipants: {
    type: Number,
    default: 0
  }
});
// Validate multiple fields together
eventSchema.pre('save', function(next) {
  
  // End date must be after start date
  if (this.endDate <= this.startDate) {
    return next(new Error('End date must be after start date'));
  }
  
  // Registered participants cannot exceed max
  if (this.registeredParticipants > this.maxParticipants) {
    return next(new Error('Registered participants cannot exceed maximum'));
  }
  
  next();
});

9. CUSTOM VALIDATOR FUNCTIONS (Reusable)

// ============================================
// 9. CUSTOM VALIDATOR FUNCTIONS (Reusable)
// ============================================
// Create reusable validator functions
const validators = {
  
  isStrongPassword: {
    validator: function(value) {
      return /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$/.test(value);
    },
    message: 'Password must be at least 8 characters with uppercase, lowercase, number, and special character'
  },
  
  isPhoneNumber: {
    validator: function(value) {
      return /^[\+]?[(]?[0-9]{3}[)]?[-\s\.]?[0-9]{3}[-\s\.]?[0-9]{4,6}$/.test(value);
    },
    message: 'Please enter a valid phone number'
  },
  
  isURL: {
    validator: function(value) {
      try {
        new URL(value);
        return true;
      } catch {
        return false;
      }
    },
    message: 'Please enter a valid URL'
  },
  
  isFutureDate: {
    validator: function(value) {
      return value > new Date();
    },
    message: 'Date must be in the future'
  }
};
// Use reusable validators
const memberSchema = new mongoose.Schema({
  phone: {
    type: String,
    validate: validators.isPhoneNumber
  },
  
  website: {
    type: String,
    validate: validators.isURL
  },
  
  expiryDate: {
    type: Date,
    validate: validators.isFutureDate
  }
});

10. HANDLING VALIDATION ERRORS

// ============================================
// 10. HANDLING VALIDATION ERRORS
// ============================================
// In your controller/route handler
const createUser = async (req, res) => {
  try {
    const user = await User.create(req.body);
    
    res.status(201).json({
      success: true,
      data: user
    });
    
  } catch (error) {
    
    // Handle validation errors
    if (error.name === 'ValidationError') {
      const errors = Object.values(error.errors).map(err => ({
        field: err.path,
        message: err.message,
        value: err.value
      }));
      
      return res.status(400).json({
        success: false,
        message: 'Validation failed',
        errors: errors
      });
    }
    
    res.status(500).json({
      success: false,
      message: error.message
    });
  }
};

🎯 Key Concepts:

  • 1. Basic Custom Validation: Use the validate property with a validator function. Return true if valid, false if invalid. Provide custom error messages.
  • 2. Access Document Fields: Use this to access other fields in the document (only works with regular functions). Validate one field based on another's value.
  • 3. Async Validation: Use async/await for database lookups. Check uniqueness or existence of related documents. Great for checking if referenced IDs exist.
  • 4. Conditional Validation: Make fields required based on conditions (using a function). Apply validation logic only when certain criteria are met.
  • 5. Array Validation: Validate array length, check each element in the array, and ensure uniqueness of array items.
  • 6. Nested Object Validation: Validate individual properties in nested objects, and validate the entire nested structure.
  • 7. Complex Password Validation: Check multiple criteria (uppercase, lowercase, numbers, special chars). Compare fields (password confirmation).
  • 8. Schema-Level Validation: Use pre('save') middleware for complex validations. Validate relationships between multiple fields (e.g., startDate and endDate).
  • 9. Reusable Validators: Create validator objects you can reuse across schemas. Keep your code DRY (Don't Repeat Yourself).
  • 10. Error Handling: Catch ValidationError specifically. Extract and format error messages nicely.

💡 Important Notes:

  • this context: In validators, this refers to the document being validated (only works with **regular functions**, not arrow functions).
  • Async validators: Automatically detected - just use async keyword.
  • Update operations: Use runValidators: true option in findByIdAndUpdate() to run validators on updates.
  • **Array of validators:** You can provide multiple validators for a single field.

🚀 Pro Tips:

  • Keep validators pure and focused on one thing.
  • Use async validators sparingly (they're slower).
  • Create reusable validator functions for common patterns.
  • Provide clear, user-friendly error messages.
  • Test your validators thoroughly.
