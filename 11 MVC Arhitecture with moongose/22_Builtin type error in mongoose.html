Deep Dive into Built-in Error Types in Mongoose

Let me teach you everything about Mongoose error types with clear explanations and practical examples.

Overview of Mongoose Error Types

Mongoose has several built-in error types that help you identify and handle different kinds of failures in your MongoDB operations. All Mongoose errors inherit from the base Error class.

1. ValidationError

Definition
Thrown when document validation fails. This occurs when you try to save a document that doesn't meet the schema validation rules.

Common Causes

  • Missing required fields
  • Data type mismatches
  • Custom validator failures
  • Enum violations
  • Min/max constraint violations

Syntax & Properties

error.name // 'ValidationError'
error.errors // Object containing individual field errors
error.message // Summary of all validation errors

2. ValidatorError

Definition
Individual field validation error. Multiple ValidatorErrors combine into a ValidationError.

Properties

error.name // 'ValidatorError'
error.path // Field name that failed
error.value // The invalid value
error.kind // Type of validation that failed
error.message // Error message

3. CastError

Definition
Thrown when Mongoose cannot convert a value to the schema's defined type.

Common Causes

  • Invalid ObjectId format
  • Wrong data type (string instead of number)
  • Invalid date formats

Properties

error.name // 'CastError'
error.path // Field that failed casting
error.value // The value that couldn't be cast
error.kind // The type it was trying to cast to

4. DocumentNotFoundError

Definition
Thrown when findOneAndUpdate, findOneAndDelete, or similar methods with {orFail: true} option don't find a document.

Properties

error.name // 'DocumentNotFoundError'
error.filter // The query filter used

5. MissingSchemaError

Definition
Thrown when you try to access a model that hasn't been registered.

6. VersionError

Definition
Thrown when saving a document fails due to version conflicts (optimistic locking).

Properties

error.name // 'VersionError'
error.version // Expected version

7. ParallelSaveError

Definition
Thrown when you call save() multiple times on the same document in parallel.

8. OverwriteModelError

Definition
Thrown when trying to register a model with a name that's already been used.

// Complete Mongoose Error Types Demo

const mongoose = require('mongoose');

// ============================================
// SCHEMA DEFINITION WITH VALIDATION
// ============================================

const userSchema = new mongoose.Schema({
  username: {
    type: String,
    required: [true, 'Username is required'],
    minlength: [3, 'Username must be at least 3 characters'],
    maxlength: [20, 'Username cannot exceed 20 characters'],
    unique: true,
    trim: true
  },
  email: {
    type: String,
    required: [true, 'Email is required'],
    unique: true,
    lowercase: true,
    validate: {
      validator: function(v) {
        return /^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$/.test(v);
      },
      message: props => `${props.value} is not a valid email!`
    }
  },
  age: {
    type: Number,
    required: true,
    min: [18, 'Must be at least 18 years old'],
    max: [100, 'Age cannot exceed 100']
  },
  role: {
    type: String,
    enum: {
      values: ['user', 'admin', 'moderator'],
      message: '{VALUE} is not a valid role'
    },
    default: 'user'
  },
  createdAt: {
    type: Date,
    default: Date.now
  }
}, {
  timestamps: true,
  versionKey: true // Enable versioning for VersionError demo
});

const User = mongoose.model('User', userSchema);

async function demonstrateValidationError() {
  console.log('--- 1. VALIDATION ERROR ---\n');
  
  try {
    const user = new User({
      username: 'ab', // Too short (minlength: 3)
      email: 'invalid-email', // Invalid email format
      age: 15, // Below minimum (min: 18)
      role: 'superuser' // Not in enum
    });
    
    await user.save();
  } catch (error) {
    if (error.name === 'ValidationError') {
      console.log('Error Name:', error.name);
      console.log('Error Message:', error.message);
      console.log('\nIndividual Field Errors:');
      
      // Loop through each validation error
      for (let field in error.errors) {
        const fieldError = error.errors[field];
        console.log(`\n  Field: ${field}`);
        console.log(`  Kind: ${fieldError.kind}`);
        console.log(`  Value: ${fieldError.value}`);
        console.log(`  Message: ${fieldError.message}`);
      }
    }
  }
  console.log('\n');
}

async function demonstrateCastError() {
  console.log('--- 2. CAST ERROR ---\n');
  
  try {
    // Try to find user with invalid ObjectId format
    const user = await User.findById('invalid-id-format');
  } catch (error) {
    if (error.name === 'CastError') {
      console.log('Error Name:', error.name);
      console.log('Path:', error.path); // '_id'
      console.log('Value:', error.value); // 'invalid-id-format'
      console.log('Kind:', error.kind); // 'ObjectId'
      console.log('Message:', error.message);
    }
  }
  
  try {
    // Try to save with wrong type
    const user = new User({
      username: 'validuser',
      email: 'valid@email.com',
      age: 'twenty-five', // String instead of Number
      role: 'user'
    });
    
    await user.save();
  } catch (error) {
    if (error.name === 'ValidationError') {
      const ageError = error.errors.age;
      console.log('\nAge Field Cast Error:');
      console.log('Kind:', ageError.kind); // 'Number'
      console.log('Value:', ageError.value); // 'twenty-five'
      console.log('Message:', ageError.message);
    }
  }
  console.log('\n');
}

// 3. DOCUMENT NOT FOUND ERROR
// ============================================

async function demonstrateDocumentNotFoundError() {
  console.log('--- 3. DOCUMENT NOT FOUND ERROR ---\n');
  
  try {
    // Use orFail() to throw error if document not found
    const user = await User.findOneAndUpdate(
      { username: 'nonexistent-user' },
      { age: 25 }
    ).orFail();
    
  } catch (error) {
    if (error.name === 'DocumentNotFoundError') {
      console.log('Error Name:', error.name);
      console.log('Filter:', JSON.stringify(error.filter));
      console.log('Message:', error.message);
    }
  }
  
  try {
    // Alternative: findByIdAndUpdate with orFail
    const fakeId = new mongoose.Types.ObjectId();
    await User.findByIdAndUpdate(fakeId, { age: 30 }).orFail();
    
  } catch (error) {
    if (error.name === 'DocumentNotFoundError') {
      console.log('\nFindByIdAndUpdate Error:');
      console.log('Message:', error.message);
    }
  }
  console.log('\n');
}

// 4. VERSION ERROR (Optimistic Locking)
// ============================================

async function demonstrateVersionError() {
  console.log('--- 4. VERSION ERROR ---\n');
  
  try {
    // First, create a user
    const user = new User({
      username: 'versiontest',
      email: 'version@test.com',
      age: 25,
      role: 'user'
    });
    await user.save();
    console.log('User created with version:', user.__v);
    
    // Fetch the same user twice
    const userCopy1 = await User.findById(user._id);
    const userCopy2 = await User.findById(user._id);
    
    // Modify and save first copy
    userCopy1.age = 26;
    await userCopy1.save();
    console.log('First save successful, new version:', userCopy1.__v);
    
    // Try to save second copy (will fail because version changed)
    userCopy2.age = 27;
    await userCopy2.save();
    
  } catch (error) {
    if (error.name === 'VersionError') {
      console.log('\nError Name:', error.name);
      console.log('Message:', error.message);
      console.log('Expected Version:', error.version);
    }
  }
  console.log('\n');
}

async function demonstrateDuplicateKeyError() {
  console.log('--- 8. DUPLICATE KEY ERROR (MongoDB) ---\n');
  
  try {
    // Create first user
    const user1 = new User({
      username: 'uniqueuser',
      email: 'unique@test.com',
      age: 25,
      role: 'user'
    });
    await user1.save();
    console.log('First user created successfully');
    
    // Try to create another user with same username (unique field)
    const user2 = new User({
      username: 'uniqueuser', // Duplicate!
      email: 'another@test.com',
      age: 30,
      role: 'user'
    });
    await user2.save();
    
  } catch (error) {
    // This is a MongoDB error, not a Mongoose error
    if (error.code === 11000) {
      console.log('\nError Code:', error.code);
      console.log('Error Name:', error.name); // MongoServerError
      console.log('Message:', error.message);
      console.log('Duplicate Key:', JSON.stringify(error.keyValue));
    }
  }
  console.log('\n');
}

// COMPREHENSIVE ERROR HANDLER
// ============================================

function handleMongooseError(error) {
  console.log('--- HANDLING ERROR ---');
  
  switch (error.name) {
    case 'ValidationError':
      console.log('Validation failed:');
      const errors = {};
      for (let field in error.errors) {
        errors[field] = error.errors[field].message;
      }
      return { status: 400, errors };
      
    case 'CastError':
      console.log('Invalid data type:');
      return {
        status: 400,
        message: `Invalid ${error.kind} for field ${error.path}`
      };
      
    case 'DocumentNotFoundError':
      console.log('Document not found:');
      return {
        status: 404,
        message: 'Resource not found'
      };
      
    case 'VersionError':
      console.log('Version conflict:');
      return {
        status: 409,
        message: 'Document was modified by another process'
      };
      
    case 'ParallelSaveError':
      console.log('Parallel save detected:');
      return {
        status: 500,
        message: 'Cannot save document multiple times simultaneously'
      };
      
    case 'MissingSchemaError':
      console.log('Schema not found:');
      return {
        status: 500,
        message: 'Internal server error: Schema not registered'
      };
      
    case 'OverwriteModelError':
      console.log('Model already exists:');
      return {
        status: 500,
        message: 'Internal server error: Duplicate model registration'
      };
      
    default:
      // Handle MongoDB errors
      if (error.code === 11000) {
        console.log('Duplicate key error:');
        return {
          status: 409,
          message: 'Duplicate value for unique field',
          field: Object.keys(error.keyValue)[0]
        };
      }
      
      console.log('Unknown error:');
      return {
        status: 500,
        message: 'An unexpected error occurred'
      };
  }
}

// EXPRESS ERROR MIDDLEWARE EXAMPLE
// ============================================

function expressErrorMiddleware(error, req, res, next) {
  console.error('Error occurred:', error);
  
  if (error.name === 'ValidationError') {
    const errors = {};
    for (let field in error.errors) {
      errors[field] = error.errors[field].message;
    }
    return res.status(400).json({
      success: false,
      message: 'Validation failed',
      errors
    });
  }
  
  if (error.name === 'CastError') {
    return res.status(400).json({
      success: false,
      message: `Invalid ${error.kind} for ${error.path}`
    });
  }
  
  if (error.name === 'DocumentNotFoundError') {
    return res.status(404).json({
      success: false,
      message: 'Resource not found'
    });
  }
  
  if (error.code === 11000) {
    return res.status(409).json({
      success: false,
      message: 'Duplicate key error',
      field: Object.keys(error.keyValue)[0]
    });
  }
  
  // Default error
  res.status(500).json({
    success: false,
    message: 'Internal server error'
  });
}

Best Practices for Error Handling

1. Always Use Try-Catch with Async/Await

async function createUser(userData) {
  try {
    const user = new User(userData);
    await user.save();
    return { success: true, user };
  } catch (error) {
    return { success: false, error: handleMongooseError(error) };
  }
}

2. Use Error Middleware in Express

app.use((error, req, res, next) => {
  if (error.name === 'ValidationError') {
    // Handle validation errors
  }
  // ... other error handling
});

3. Validate Before Saving

const user = new User(userData);
const validationError = user.validateSync(); // Synchronous validation
if (validationError) {
  // Handle errors without saving to DB
}

4. Use Custom Error Messages

const schema = new Schema({
  email: {
    type: String,
    required: [true, 'Please provide an email address'],
    validate: {
      validator: v => /regex/.test(v),
      message: 'Please provide a valid email'
    }
  }
});
