  2. Establishing Connection

    // db.js - Database connection file

    const mongoose = require('mongoose');

// Connection string format: mongodb://localhost:27017/database_name
const connectDB = async () => {
  try {
    const conn = await mongoose.connect('mongodb://localhost:27017/myapp', {
      useNewUrlParser: true,
      useUnifiedTopology: true,
    });
        
    console.log(`MongoDB Connected: ${conn.connection.host}`);
  } catch (error) {
    console.error(`Error: ${error.message}`);
    process.exit(1); // Exit with failure
  }
};

// Connection events
mongoose.connection.on('connected', () => {
  console.log('Mongoose connected to DB');
});

mongoose.connection.on('error', (err) => {
  console.log('Mongoose connection error:', err);
});

mongoose.connection.on('disconnected', () => {
  console.log('Mongoose disconnected');
});

// Graceful shutdown
process.on('SIGINT', async () => {
  await mongoose.connection.close();
  console.log('Mongoose connection closed due to app termination');
  process.exit(0);
});

module.exports = connectDB;

  3. Defining Schemas and Models

    Mongoose uses Schemas to define the structure of documents and Models to interact with the database.

    // models/User.js

    const mongoose = require('mongoose');

// Define Schema
const userSchema = new mongoose.Schema({
  // Basic field types
  name: {
    type: String,
    required: [true, 'Name is required'], // Custom error message
    trim: true, // Removes whitespace
    minlength: [3, 'Name must be at least 3 characters'],
    maxlength: [50, 'Name cannot exceed 50 characters']
  },
  
  email: {
    type: String,
    required: true,
    unique: true, // Creates unique index
    lowercase: true, // Converts to lowercase
    match: [/^\S+@\S+\.\S+$/, 'Please enter a valid email']
  },
  
  age: {
    type: Number,
    min: [0, 'Age cannot be negative'],
    max: [150, 'Age seems unrealistic'],
    default: 18
  },
  
  isActive: {
    type: Boolean,
    default: true
  },
  
  // Array of strings
  hobbies: [String],
  
  // Array of objects
  addresses: [{
    street: String,
    city: String,
    zipCode: String,
    isPrimary: {
      type: Boolean,
      default: false
    }
  }],
  
  // Nested object
  profile: {
    bio: String,
    website: String,
    social: {
      twitter: String,
      linkedin: String
    }
  },
  
  // Reference to another model
  posts: [{
    type: mongoose.Schema.Types.ObjectId,
    ref: 'Post' // Reference to Post model
  }],
  
  // Date fields
  registeredAt: {
    type: Date,
    default: Date.now
  },
  
  lastLogin: Date,
  
  // Enum - restrict to specific values
  role: {
    type: String,
    enum: ['user', 'admin', 'moderator'],
    default: 'user'
  },
  
  // Mixed type - can store any type
  metadata: mongoose.Schema.Types.Mixed

}, {
  // Schema options
  timestamps: true, // Adds createdAt and updatedAt automatically
  versionKey: false // Removes __v field
});

// Virtual property (not stored in DB)
userSchema.virtual('fullInfo').get(function() {
  return `${this.name} (${this.email})`;
});

// Instance method
userSchema.methods.getPublicProfile = function() {
  return {
    name: this.name,
    email: this.email,
    role: this.role
  };
};

// Static method
userSchema.statics.findByEmail = function(email) {
  return this.findOne({ email: email });
};

// Pre-save middleware (runs before saving)
userSchema.pre('save', function(next) {
  console.log('About to save user:', this.name);
  next();
});

// Post-save middleware (runs after saving)
userSchema.post('save', function(doc, next) {
  console.log('User saved:', doc.name);
  next();
});

// Create and export model
const User = mongoose.model('User', userSchema);

module.exports = User;

// METHOD 4: Create with validation handling
// ============================================
const createUserWithValidation = async (userData) => {
  try {
    const user = new User(userData);
    
    // Validate before saving
    const validationError = user.validateSync();
    if (validationError) {
      console.log('Validation errors:', validationError.errors);
      return null;
    }
    
    await user.save();
    console.log('User created successfully');
    return user;
  } catch (error) {
    if (error.name === 'ValidationError') {
      // Handle validation errors
      Object.keys(error.errors).forEach(key => {
        console.log(`${key: ${error.errors[key].message}`);
      });
    } else if (error.code === 11000) {
      // Handle duplicate key error
      console.log('Email already exists');
    } else {
      console.error('Error:', error.message);
    }
  }
};

const queryOperators = async () => {
  try {
    // Comparison operators
    const ageQuery = await User.find({
      age: { 
        $gt: 18,   // Greater than
        $gte: 18,  // Greater than or equal
        $lt: 65,   // Less than
        $lte: 65,  // Less than or equal
        $ne: 30,   // Not equal
        $in: [25, 30, 35],  // In array
        $nin: [40, 45]     // Not in array
      }
    });
    
    // Logical operators
    const logicalQuery = await User.find({
      $and: [
        { age: { $gte: 18 } },
        { isActive: true }
      ]
    });
    
    const orQuery = await User.find({
      $or: [
        { role: 'admin' },
        { age: { $gte: 60 } }
      ]
    });
    
    const notQuery = await User.find({
      age: { $not: { $gte: 18 } }
    });
    
    // Element operators
    const existsQuery = await User.find({
      lastLogin: { $exists: true }  // Field exists
    });
    
    const typeQuery = await User.find({
      age: { $type: 'number' }  // Field type
    });
    
    // Array operators
    const arrayQuery = await User.find({
      hobbies: { $size: 2 }  // Array size
    });
    
    const allQuery = await User.find({
      hobbies: { $all: ['reading', 'gaming'] }  // Contains all
    });
    
    const elemMatchQuery = await User.find({
      addresses: { 
        $elemMatch: { 
          city: 'New York',
          isPrimary: true 
        }
      }
    });
    
    // String operators (Regex)
    const regexQuery = await User.find({
      name: { $regex: /john/i }  // Case-insensitive search
    });
    
    return { ageQuery, logicalQuery };
  } catch (error) {
    console.error('Error:', error.message);
  }

const updateOperators = async (userId) => {
  try {
    // $set - Set field value
    await User.updateOne(
      { _id: userId },
      { $set: { name: 'John', age: 30 } }
    );
    
    // $unset - Remove field
    await User.updateOne(
      { _id: userId },
      { $unset: { lastLogin: '' } }
    );
    
    // $inc - Increment/decrement number
    await User.updateOne(
      { _id: userId },
      { 
        $inc: { age: 1 },    // Increment by 1
        $inc: { score: -5 }   // Decrement by 5
      }
    );
    
    // $mul - Multiply
    await User.updateOne(
      { _id: userId },
      { $mul: { score: 2 } }
    );
    
    // $min - Update if new value is less
    await User.updateOne(
      { _id: userId },
      { $min: { age: 25 } }  // Only updates if 25 < current age
    );
    
    // $max - Update if new value is greater
    await User.updateOne(
      { _id: userId },
      { $max: { score: 100 } }
    );
    
    // $currentDate - Set to current date
    await User.updateOne(
      { _id: userId },
      { $currentDate: { lastModified: true } }
    );
    
    // $rename - Rename field
    await User.updateOne(
      { _id: userId },
      { $rename: { 'name': 'fullName' } }
    );
    
    console.log('Update operators executed');
  } catch (error) {
    console.error('Error:', error.message);
  }
};

// ARRAY UPDATE OPERATORS
// ============================================

const arrayUpdateOperators = async (userId) => {
  try {
    // $push - Add to array
    await User.updateOne(
      { _id: userId },
      { $push: { hobbies: 'swimming' } }
    );
    
    // $push with $each - Add multiple items
    await User.updateOne(
      { _id: userId },
      { 
        $push: { 
          hobbies: { 
            $each: ['reading', 'gaming', 'cooking'],
            $position: 0  // Insert at beginning
          }
        }
      }
    );
    
    // $addToSet - Add if doesn't exist (prevents duplicates)
    await User.updateOne(
      { _id: userId },
      { $addToSet: { hobbies: 'reading' } }
    );
    
    // $pull - Remove matching items
    await User.updateOne(
      { _id: userId },
      { $pull: { hobbies: 'gaming' } }
    );
    
    // $pull with condition
    await User.updateOne(
      { _id: userId },
      { 
        $pull: { 
          addresses: { city: 'New York' }
        }
      }
    );
    
    // $pop - Remove first or last element
    await User.updateOne(
      { _id: userId },
      { $pop: { hobbies: 1 } }  // 1 = last, -1 = first
    );
    
    // $pullAll - Remove multiple values
    await User.updateOne(
      { _id: userId },
      { $pullAll: { hobbies: ['reading', 'gaming'] } }
    );
    
    // $ (positional) - Update first matching array element
    await User.updateOne(
      { _id: userId, 'addresses.city': 'New York' },
      { $set: { 'addresses.$.isPrimary': true } }
    );
    
    // $[] - Update all array elements
    await User.updateOne(
      { _id: userId },
      { $set: { 'addresses.$[].isPrimary': false } }
    );
    
    // $[] - Update specific array elements
    await User.updateOne(
      { _id: userId },
      { 
        $set: { 'addresses.$[elem].isPrimary': true }
      },
      {
        arrayFilters: [{ 'elem.city': 'New York' }]
      }
    );
    
    console.log('Array update operators executed');
  } catch (error) {
    console.error('Error:', error.message);
  }
};

10. Error Handling & Best Practices


const mongoose = require('mongoose');

// ============================================
// 1. COMPREHENSIVE ERROR HANDLING
// ============================================

class DatabaseError extends Error {
  constructor(message, type, details = {}) {
    super(message);
    this.name = 'DatabaseError';
    this.type = type;
    this.details = details;
  }
}

const handleMongooseError = (error) => {
  // Validation Error
  if (error.name === 'ValidationError') {
    const errors = Object.keys(error.errors).map(key => ({
      field: key,
      message: error.errors[key].message
    }));
    
    return new DatabaseError(
      'Validation failed',
      'VALIDATION_ERROR',
      { errors }
    );
  }
  
  // Duplicate Key Error
  if (error.code === 11000) {
    const field = Object.keys(error.keyPattern)[0];
    return new DatabaseError(
      `${field} already exists`,
      'DUPLICATE_KEY',
      { field, value: error.keyValue[field] }
    );
  }
  
  // Cast Error (invalid ID format)
  if (error.name === 'CastError') {
    return new DatabaseError(
      `Invalid ${error.path}: ${error.value}`,
      'CAST_ERROR',
      { path: error.path, value: error.value }
    );
  }
  
  // Document Not Found
  if (error.name === 'DocumentNotFoundError') {
    return new DatabaseError(
      'Document not found',
      'NOT_FOUND',
      {}
    );
  }
  
  // Connection Error
  if (error.name === 'MongoNetworkError') {
    return new DatabaseError(
      'Database connection failed',
      'CONNECTION_ERROR',
      { originalError: error.message }
    );
  }
  
  // Default error
  return new DatabaseError('An unexpected database error occurred', 'UNKNOWN_ERROR', { originalError: error.message });
};

🎓 Key Concepts Explained

How Mongoose Works:

  • Connection Flow: Mongoose connects to MongoDB → Creates connection pool → Maintains persistent connections
  • Schema to Model: Schema defines structure → Model is a constructor compiled from Schema → Model creates documents
  • Document Lifecycle:
    • New document → Validation → Pre-save hooks → Save to DB → Post-save hooks
    • Query → Pre-query hooks → Execute query → Post-query hooks → Return results
  • Query Execution:
    • Queries are chainable and lazy (don't execute until .exec(), await, or callback)
    • Example: Model.find().where('age').gt(18).limit(10) builds query but doesn't execute
  • Middleware System:
    • Pre hooks run before operation
    • Post hooks run after operation
    • Can modify documents or queries
  • Validation: Happens on save() and create(), optional on update() with runValidators: true
  • Performance:
    • .lean() returns plain objects (no Mongoose methods, faster)
    • .select() reduces data transfer
    • Indexes speed up queries dramatically
  • When to Use What:
    • save(): When you need middleware hooks and full validation
    • create(): Quick creation with validation
    • updateOne(): When you only need confirmation, not the document
    • findByIdAndUpdate(): When you need the updated document returned
    • lean(): For read-only operations where you don't need Mongoose features
    • populate(): For small-medium datasets; avoid for large-scale joins
