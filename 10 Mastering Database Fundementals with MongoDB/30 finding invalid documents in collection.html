You can find invalid documents in a MongoDB collection from Node.js by using the **$jsonSchema** operator or the **db.getCollectionInfos()** command to retrieve the validation rules, and then using that rule as a **negative query** with the **$not** operator in your **find()** operation.

This method requires your collection to have **schema validation** defined.

1. Get the Validation Schema

First, you need to retrieve the actual validation rules that the collection is currently using.

You can use the Node.js MongoDB driver's **db.listCollections()** method or **db.getCollectionInfos()** to find the validation object for your collection.

JavaScript 
// Assuming 'db' is your connected MongoDB database instance
async function getValidationSchema(db, collectionName) {
    const collections = await db.listCollections({ name: collectionName }).toArray();

    if (collections.length === 0) {
        throw new Error(`Collection '${collectionName}' not found.`);
    }

    const options = collections[0].options;
    const validator = options.validator;

    if (!validator || Object.keys(validator).length === 0) {
        console.warn(`Collection '${collectionName}' has no active schema validation.`);
        return null;
    }

    // The validator object is the rule you need to invert.
    return validator;
}

2. Construct the Negative Query

Once you have the validation rule (the **validator** object), you wrap it in the **$not** logical operator to find documents that **do not** match the schema.

The final query will look like this: **{ "$not": { <your_validation_rule> } }**

JavaScript 
async function findInvalidDocuments(db, collectionName) {
    const validator = await getValidationSchema(db, collectionName);

    if (!validator) {
        return []; // Return empty array if no validation is set
    }

    // Construct the query: find documents that do NOT match the validator rule
    const invalidDocumentQuery = { 
        "$not": validator 
    };

    console.log("Searching for documents with query:", JSON.stringify(invalidDocumentQuery));

    // Execute the find operation
    const invalidDocuments = await db
        .collection(collectionName)
        .find(invalidDocumentQuery)
        .toArray();

    return invalidDocuments;
}

Complete Node.js Example (Conceptual)

Here is how you would piece it together in a full connection context:

JavaScript 
const { MongoClient } = require('mongodb');
const uri = "mongodb://localhost:27017";
const client = new MongoClient(uri);
const DB_NAME = "myAppDB";
const COLLECTION_NAME = "users";

async function main() {
    try {
        await client.connect();
        const db = client.db(DB_NAME);

        const invalidDocs = await findInvalidDocuments(db, COLLECTION_NAME);

        console.log(`\nFound ${invalidDocs.length} invalid documents in '${COLLECTION_NAME}':`);
        console.table(invalidDocs);

    } catch (error) {
        console.error("An error occurred:", error.message);
    } finally {
        await client.close();
    }
}
