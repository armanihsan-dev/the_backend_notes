Line Items (Order Items) in Lemon Squeezy

What are Line Items?

In Lemon Squeezy, **line items** (officially called **Order Items**) are individual products/variants that a customer purchases in an order. They represent each distinct item on a receipt or invoice.

Analogy:

Think of a grocery receipt:

  • **Order** = The entire purchase transaction
  • **Order Items** = Each individual product line:

🛒 ORDER #12345
├── 🥤 Lemonade (Citrus Blast)  $9.99  → Order Item 1
├── 🍪 Cookies (Chocolate Chip) $5.99  → Order Item 2
└── 🍋 Lemons (Organic)        $3.99  → Order Item 3

Each of those product lines is an **Order Item**.

Key Concepts:

1. Order vs Order Item Relationship


// One Order can have multiple Order Items
Order {
  id: 1,
  total: $19.97,
  order_items: [
    { product: "Lemonade", variant: "Citrus Blast", price: $9.99, quantity: 1 },
    { product: "Cookies", variant: "Chocolate Chip", price: $5.99, quantity: 1 },
    { product: "Lemons", variant: "Organic", price: $3.99, quantity: 1 }
  ]
}

2. Order Item Attributes Explained

From your example:


{
  "order_id": 1, // Which order this belongs to
  "product_id": 1, // Reference to the Product
  "variant_id": 1, // Reference to the Variant
  "product_name": "Lemonade", // Name at time of purchase
  "variant_name": "Citrus Blast", // Variant name at time of purchase
  "price": 999, // Price in cents ($9.99)
  "quantity": 1, // How many were purchased
  "created_at": "...", // When the item was added
  "updated_at": "..." // Last update timestamp
}

**Important:** `product_name` and `variant_name` are **snapshots** - they preserve what the customer saw at purchase time, even if you later rename your products.

API Operations with Examples:

1. Retrieve a Single Order Item


# GET a specific order item
curl "https://api.lemonsqueezy.com/v1/order-items/1" \
  -H 'Accept: application/vnd.api+json' \
  -H 'Authorization: Bearer YOUR_API_KEY'

Node.js Implementation:


const https = require('https');

function getOrderItem(orderItemId) {
  return new Promise((resolve, reject) => {
    const options = {
      headers: {
        'Accept': 'application/vnd.api+json',
        'Authorization': `Bearer ${process.env.LEMON_API_KEY}`
      }
    };

    const req = https.get(
      `https://api.lemonsqueezy.com/v1/order-items/${orderItemId}`,
      options,
      (res) => {
        let data = '';
        res.on('data', chunk => data += chunk);
        res.on('end', () => {
          const response = JSON.parse(data);
          resolve(response.data);
        });
      }
    );

    req.on('error', reject);
  });
}

// Usage
getOrderItem(1).then(item => {
  console.log(`Product: ${item.attributes.product_name}`);
  console.log(`Price: $${(item.attributes.price / 100).toFixed(2)}`);
  console.log(`Quantity: ${item.attributes.quantity}`);
});

2. List All Order Items (with filtering)


# GET all order items with pagination
curl "https://api.lemonsqueezy.com/v1/order-items?page[number]=1&page[size]=10" \
  -H 'Accept: application/vnd.api+json' \
  -H 'Authorization: Bearer YOUR_API_KEY'

# Filter by order_id
curl "https://api.lemonsqueezy.com/v1/order-items?filter[order_id]=123" \
  -H 'Accept: application/vnd.api+json' \
  -H 'Authorization: Bearer YOUR_API_KEY'

# Filter by product_id
curl "https://api.lemonsqueezy.com/v1/order-items?filter[product_id]=456" \
  -H 'Accept: application/vnd.api+json' \
  -H 'Authorization: Bearer YOUR_API_KEY'

Node.js Implementation with Filtering:


async function listOrderItems(filters = {}) {
  const apiKey = process.env.LEMON_API_KEY;

  // Build query parameters
  const params = new URLSearchParams();

  // Add filters
  if (filters.orderId) {
    params.append('filter[order_id]', filters.orderId);
  }
  if (filters.productId) {
    params.append('filter[product_id]', filters.productId);
  }

  // Pagination
  params.append('page[number]', filters.page || 1);
  params.append('page[size]', filters.limit || 10);

  const url = `https://api.lemonsqueezy.com/v1/order-items?${params}`;

  const response = await fetch(url, {
    headers: {
      'Accept': 'application/vnd.api+json',
      'Authorization': `Bearer ${apiKey}`
    }
  });

  return await response.json();
}

// Usage examples
const allItems = await listOrderItems({ limit: 50 });
const orderItems = await listOrderItems({ orderId: 123 });
const productItems = await listOrderItems({ productId: 456, page: 2 });

3. Get Order Items for a Specific Order


async function getOrderItemsByOrder(orderId) {
  // Method 1: Using filter on order-items endpoint
  const items = await listOrderItems({ orderId });

  // Method 2: Using relationships (if you have the order data)
  // First get the order, then follow the relationship
  const orderResponse = await fetch(
    `https://api.lemonsqueezy.com/v1/orders/${orderId}?include=order-items`,
    {
      headers: {
        'Accept': 'application/vnd.api+json',
        'Authorization': `Bearer ${process.env.LEMON_API_KEY}`
      }
    }
  );

  const orderData = await orderResponse.json();
  // Includes order items in the response
  return orderData.included;
}

Practical Use Cases:

1. Generate Detailed Invoice


async function generateDetailedInvoice(orderId) {
  // Get order details
  const orderRes = await fetch(`https://api.lemonsqueezy.com/v1/orders/${orderId}`, {
    headers: { /* auth headers */ }
  });
  const order = await orderRes.json();

  // Get all items for this order
  const items = await listOrderItems({ orderId });

  // Build invoice lines
  const invoiceLines = items.data.map(item => {
    const attrs = item.attributes;
    return {
      description: `${attrs.product_name} - ${attrs.variant_name}`,
      quantity: attrs.quantity,
      unitPrice: attrs.price / 100,
      total: (attrs.price * attrs.quantity) / 100
    };
  });

  return {
    orderNumber: order.data.attributes.identifier,
    date: order.data.attributes.created_at,
    customer: order.data.attributes.customer_name,
    items: invoiceLines,
    total: order.data.attributes.total / 100
  };
}

2. Calculate Product Revenue


async function getProductRevenue(productId) {
  // Get all orders for this product
  const items = await listOrderItems({ productId });

  let totalRevenue = 0;
  let totalQuantity = 0;

  items.data.forEach(item => {
    const attrs = item.attributes;
    totalRevenue += (attrs.price * attrs.quantity);
    totalQuantity += attrs.quantity;
  });

  return {
    productId,
    totalRevenue: totalRevenue / 100,
    totalQuantity,
    averagePrice: (totalRevenue / totalQuantity) / 100
  };
}

3. Get Top-Selling Products


async function getTopSellingProducts(limit = 10) {
  // Get all order items
  const allItems = await listOrderItems({ limit: 1000 }); // Adjust as needed

  // Aggregate by product
  const productSales = {};

  allItems.data.forEach(item => {
    const attrs = item.attributes;
    const productId = attrs.product_id;

    if (!productSales[productId]) {
      productSales[productId] = {
        productName: attrs.product_name,
        totalQuantity: 0,
        totalRevenue: 0
      };
    }

    productSales[productId].totalQuantity += attrs.quantity;
    productSales[productId].totalRevenue += (attrs.price * attrs.quantity);
  });

  // Sort and return top products
  return Object.entries(productSales)
    .map(([id, data]) => ({
      productId: id,
      ...data,
      totalRevenue: data.totalRevenue / 100
    }))
    .sort((a, b) => b.totalQuantity - a.totalQuantity)
    .slice(0, limit);
}

Important Notes:

  • **Price Format:** Always divide by 100 (prices are in cents)
  • **Data Immutability:** Order items preserve historical data
  • **Relationships:** Use include parameter to get related data
  • **Pagination:** Always implement pagination for lists
  • **Filtering:** Use filters to avoid fetching unnecessary data
