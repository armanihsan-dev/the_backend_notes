Complete Guide to ngrok

1. 🔍 What is ngrok?

Simple Definition

ngrok is a reverse proxy tool that creates a secure tunnel from a public endpoint to your local development server.

In Plain English

Imagine your local computer (localhost) is a private island, and the internet is the ocean. ngrok builds a bridge/tunnel that lets boats (internet traffic) reach your island without you having to deploy it to a public cloud server.

Key Features

  • Creates public URLs for local servers
  • HTTPS support even for HTTP local servers
  • Request inspection (see incoming/outgoing traffic)
  • Authentication options
  • Traffic replay capabilities
  • Web interface for monitoring

2. ⚙️ How ngrok Works

Technical Architecture

// How data flows through ngrok:
1. User visits → https://your-subdomain.ngrok.io
   ↓
2. Request goes to → ngrok's cloud servers
   ↓
3. ngrok forwards to → Secure tunnel → Your computer
   ↓
4. Your app processes → http://localhost:3000
   ↓
5. Response goes back through → Same tunnel in reverse

The Tunnel Mechanism

# 1. You run local server
node server.js  # Listening on http://localhost:3000

# 2. You start ngrok
ngrok http 3000

# 3. ngrok creates this tunnel:
Local: http://localhost:3000  ←→  Public: https://abc123.ngrok.io

Why It's Special

  • No firewall configuration needed
  • No port forwarding required
  • Works behind NAT (home/office networks)
  • SSL certificates automatically provided
  • Subdomains automatically assigned

6. 🧪 Testing Webhooks with Express.js

6.1 Complete Express.js Webhook Server

// server.js - Complete webhook server for testing
const express = require('express');
const crypto = require('crypto');
require('dotenv').config();

const app = express();
const PORT = process.env.PORT || 3000;

// Middleware to parse JSON
app.use(express.json());

// Webhook signature verification middleware
const verifyWebhookSignature = (req, res, next) => {
  try {
    const signature = req.headers['x-signature'];
    const secret = process.env.LEMON_SQUEEZY_WEBHOOK_SECRET;
    
    if (!signature || !secret) {
      console.warn('⚠️ Missing signature or secret');
      return res.status(401).json({ error: 'Unauthorized' });
    }
    
    // Create HMAC signature
    const hmac = crypto.createHmac('sha256', secret);
    const digest = hmac.update(JSON.stringify(req.body)).digest('hex');
    
    // Compare signatures
    if (signature !== digest) {
      console.error('❌ Invalid webhook signature');
      return res.status(401).json({ error: 'Invalid signature' });
    }
    
    console.log('✅ Webhook signature verified');
    next();
  } catch (error) {
    console.error('❌ Signature verification error:', error);
    res.status(500).json({ error: 'Verification failed' });
  }
};

// Webhook endpoint
app.post('/api/webhooks/lemon', verifyWebhookSignature, (req, res) => {
  try {
    const event = req.body;
    const eventName = event.meta.event_name;
    
    console.log('='.repeat(50));
    console.log(`📨 WEBHOOK RECEIVED: ${eventName}`);
    console.log('='.repeat(50));
    
    // Log full event for inspection
    console.log('📋 Event Details:');
    console.log(JSON.stringify(event, null, 2));
    
    // Process different event types
    switch (eventName) {
      case 'order_created':
        console.log('🛒 New order created!');
        console.log(`Order ID: ${event.data.id}`);
        console.log(`Customer: ${event.data.attributes.user_email}`);
        console.log(`Amount: $${event.data.attributes.total / 100}`);
        break;
        
      case 'subscription_created':
        console.log('🔔 New subscription created!');
        console.log(`Subscription ID: ${event.data.id}`);
        console.log(`Plan: ${event.data.attributes.variant_name}`);
        console.log(`Status: ${event.data.attributes.status}`);
        break;
        
      case 'subscription_cancelled':
        console.log('🚫 Subscription cancelled');
        console.log(`Customer: ${event.data.attributes.user_email}`);
        console.log(`Cancelled at: ${event.data.attributes.ends_at}`);
        break;
        
      default:
        console.log(`⚠️ Unhandled event type: ${eventName}`);
    }
    
    // ALWAYS return 200 immediately
    console.log('✅ Webhook processed successfully');
    res.status(200).json({ received: true });
    
  } catch (error) {
    console.error('❌ Webhook processing error:', error);
    // Still return 200 to prevent retries
    res.status(200).json({ received: true, error: error.message });
  }
});

// Test endpoint to verify server is running
app.get('/api/test', (req, res) => {
  res.json({
    message: 'Webhook server is running!',
    timestamp: new Date().toISOString(),
    endpoints: {
      webhook: 'POST /api/webhooks/lemon',
      test: 'GET /api/test'
    }
  });
});

// Root endpoint
app.get('/', (req, res) => {
  res.send(`
    
      Webhook Test Server
      
        
🎯 Webhook Test Server

        
Server is running on port ${PORT}

        
Endpoints:

        
    
          
  • Webhook: POST /api/webhooks/lemon
    • 
          
  • Test: GET /api/test
    • 
        

        
Use ngrok to get a public URL for webhook testing!

      
    
  `);
});

// Start server
app.listen(PORT, () => {
  console.log('='.repeat(50));
  console.log(`🚀 Server running on http://localhost:${PORT}`);
  console.log(`📬 Webhook endpoint: http://localhost:${PORT}/api/webhooks/lemon`);
  console.log(`🧪 Test endpoint: http://localhost:${PORT}/api/test`);
  console.log('='.repeat(50));
  console.log('\n💡 To make it public, run: ngrok http ' + PORT);
});

7.2 Using ngrok Programmatically (Node.js)

// ngrok-programmatic.js
const ngrok = require('ngrok');
const express = require('express');

async function startServerWithNgrok() {
  const app = express();
  const PORT = 3000;
  
  app.use(express.json());
  
  app.post('/webhook', (req, res) => {
    console.log('Webhook received:', req.body);
    res.sendStatus(200);
  });
  
  // Start Express server
  const server = app.listen(PORT, () => {
    console.log(`✅ Local server started on port ${PORT}`);
  });
  
  // Start ngrok tunnel
  try {
    const url = await ngrok.connect({
      addr: PORT,
      subdomain: 'mywebhooktest', // Optional custom subdomain
      authtoken: process.env.NGROK_AUTHTOKEN,
      region: 'us', // us, eu, ap, au, sa, jp, in
      onStatusChange: (status) => {
        console.log(`📡 Ngrok status: ${status}`);
      },
      onLogEvent: (data) => {
        console.log('📝 Ngrok log:', data);
      }
    });
    
    console.log('='.repeat(50));
    console.log(`🚀 PUBLIC URL: ${url}`);
    console.log(`📬 Webhook endpoint: ${url}/webhook`);
    console.log('='.repeat(50));
    console.log('\n💡 Add this URL to Lemon Squeezy webhook configuration');
    
    // Graceful shutdown
    process.on('SIGINT', async () => {
      console.log('\n👋 Shutting down...');
      await ngrok.disconnect();
      await ngrok.kill();
      server.close();
      process.exit(0);
    });
    
  } catch (error) {
    console.error('❌ Failed to start ngrok:', error);
    server.close();
  }
}

8. 🔒 Security Considerations

8.1 Is ngrok Secure for Webhook Testing?

YES, with proper configuration:

// SECURE PRACTICES:

// 1. Always verify webhook signatures
const verifySignature = (signature, payload, secret) => {
  const hmac = crypto.createHmac('sha256', secret);
  return signature === hmac.update(JSON.stringify(payload)).digest('hex');
};

// 2. Use environment variables for secrets
// NEVER hardcode in your code

// 3. ngrok provides HTTPS automatically
// Even though your local server is HTTP, ngrok tunnel is HTTPS

// 4. Free ngrok URLs are random and hard to guess
// https://abc123-def456.ngrok.io

8.2 Security Risks & Mitigation

# Potential risks:
1. **Temporary public exposure** - Your local server is public
2. **Random subdomains** - Could be guessed (low probability)
3. **No authentication** - Anyone with the URL can access

# Mitigations:
1. **Use webhook signatures** - Mandatory for Lemon Squeezy
2. **Add basic auth** - ngrok --basic-auth flag
3. **Custom subdomains** - Paid feature, harder to guess
4. **Whitelist IPs** - If your local app supports it
5. **Short-lived testing** - Don't leave ngrok running indefinitely

8.3 Production vs Development

// 🚫 NEVER use ngrok in production
// Production should use:
// 1. Real domain with SSL certificate

// 3. Production-grade security
// 4. Monitoring and logging

9. 🎯 Real Webhook Testing Workflow

Complete Testing Script

// test-webhook-workflow.js
const axios = require('axios');
const crypto = require('crypto');

class WebhookTester {
  constructor(ngrokUrl, webhookSecret) {
    this.ngrokUrl = ngrokUrl;
    this.secret = webhookSecret;
    this.endpoint = `${ngrokUrl}/api/webhooks/lemon`;
  }
  
  // Generate valid signature for testing
  generateSignature(payload) {
    const hmac = crypto.createHmac('sha256', this.secret);
    return hmac.update(JSON.stringify(payload)).digest('hex');
  }
  
  // Test different webhook events
  async testOrderCreated() {
    const payload = {
      meta: {
        event_name: "order_created",
        custom_data: { user_id: "test_123" },
        test_mode: true
      },
      data: {
        type: "orders",
        id: `test_order_${Date.now()}`,
        attributes: {
          store_id: 1,
          user_email: "test.customer@example.com",
          identifier: `order_${Date.now()}`,
          total: 2900, // $29.00 in cents
          currency: "USD",
          status: "paid",
          created_at: new Date().toISOString(),
          urls: {
            receipt: "https://app.lemonsqueezy.com/my-orders/..."
          }
        }
      }
    };
    
    const signature = this.generateSignature(payload);
    
    try {
      const response = await axios.post(this.endpoint, payload, {
        headers: {
          'x-signature': signature,
          'content-type': 'application/json'
        }
      });
      
      console.log('✅ Test order_created: SUCCESS');
      console.log('Response:', response.data);
      return true;
    } catch (error) {
      console.error('❌ Test order_created: FAILED');
      console.error('Error:', error.response?.data || error.message);
      return false;
    }
  }
  
  async testSubscriptionCreated() {
    const payload = {
      meta: {
        event_name: "subscription_created",
        test_mode: true
      },
      data: {
        type: "subscriptions",
        id: `test_sub_${Date.now()}`,
        attributes: {
          store_id: 1,
          customer_id: 123,
          order_id: 456,
          product_id: 789,
          variant_id: 101112,
          product_name: "Pro Plan",
          variant_name: "Pro Monthly",
          user_name: "Test Customer",
          user_email: "test.customer@example.com",
          status: "active",
          status_formatted: "Active",
          renews_at: new Date(Date.now() + 30*24*60*60*1000).toISOString(),
          ends_at: null,
          trial_ends_at: null,
          created_at: new Date().toISOString(),
          updated_at: new Date().toISOString()
        }
      }
    };
    
    const signature = this.generateSignature(payload);
    
    try {
      const response = await axios.post(this.endpoint, payload, {
        headers: {
          'x-signature': signature,
          'content-type': 'application/json'
        }
      });
      
      console.log('✅ Test subscription_created: SUCCESS');
      return true;
    } catch (error) {
      console.error('❌ Test subscription_created: FAILED');
      console.error('Error:', error.response?.data || error.message);
      return false;
    }
  }
  
  // Run all tests
  async runAllTests() {
    console.log('🧪 Starting webhook tests...');
    console.log(`📡 Testing endpoint: ${this.endpoint}`);
    console.log('='.repeat(50));
    
    const tests = [
      { name: 'Order Created', fn: () => this.testOrderCreated() },
      { name: 'Subscription Created', fn: () => this.testSubscriptionCreated() }
    ];
    
    let passed = 0;
    let failed = 0;
    
    for (const test of tests) {
      console.log(`\n🔍 Testing: ${test.name}`);
      const success = await test.fn();
      
      if (success) {
        passed++;
      } else {
        failed++;
      }
      
      // Small delay between tests
      await new Promise(resolve => setTimeout(resolve, 1000));
    }
    
    console.log('\n' + '='.repeat(50));
    console.log('📊 TEST SUMMARY');
    console.log('='.repeat(50));
    console.log(`✅ Passed: ${passed}`);
    console.log(`❌ Failed: ${failed}`);
    console.log(`📈 Success Rate: ${Math.round((passed/tests.length)*100)}%`);
    
    return failed === 0;
  }
}

// Usage
const tester = new WebhookTester(
  'https://abc123.ngrok.io',
  process.env.LEMON_SQUEEZY_WEBHOOK_SECRET
);

tester.runAllTests().then(success => {
  process.exit(success ? 0 : 1);
});
