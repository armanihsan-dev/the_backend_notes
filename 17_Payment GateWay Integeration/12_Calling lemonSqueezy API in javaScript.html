Lemon Squeezy API Integration Notes

1. First, Get Your API Key

  • Go to your **Lemon Squeezy dashboard**
  • Navigate to **Settings → API**
  • Create a new API key with appropriate permissions

2. Basic Setup

javascript 
// config.js - Keep your API key secure (don't commit to public repos!)
const LEMON_SQUEEZY_API_KEY = 'your_api_key_here';
const API_BASE_URL = 'https://api.lemonsqueezy.com/v1';

// Common headers for all requests
const headers = {
  'Accept': 'application/vnd.api+json',
  'Content-Type': 'application/vnd.api+json',
  'Authorization': `Bearer ${LEMON_SQUEEZY_API_KEY}`
};

Example 1: List All Stores

javascript 
// Example 1: List all stores
async function listStores() {
  try {
    const response = await fetch(`${API_BASE_URL}/stores`, {
      method: 'GET',
      headers: headers
    });

    if (!response.ok) {
      throw new Error(`HTTP error! Status: ${response.status}`);
    }

    const data = await response.json();
    
    console.log('Total stores:', data.meta.page.total);
    
    // Display store information
    data.data.forEach(store => {
      console.log(`
        Store ID: ${store.id}
        Name: ${store.attributes.name}
        URL: ${store.attributes.url}
        Created: ${new Date(store.attributes.created_at).toLocaleDateString()}
      `);
    });
    
    return data.data;
  } catch (error) {
    console.error('Error fetching stores:', error);
    throw error;
  }
}

// Usage
listStores()
  .then(stores => {
    console.log('Fetched stores:', stores);
  })
  .catch(error => {
    console.error('Failed to fetch stores:', error);
  });

Example 2: List Products with Pagination

javascript 
// Example 2: List products with pagination support
async function listProducts(storeId, page = 1) {
  try {
    const url = new URL(`${API_BASE_URL}/products`);
    
    // Add query parameters
    url.searchParams.append('page[number]', page);
    url.searchParams.append('page[size]', 10); // 10 per page
    
    if (storeId) {
      url.searchParams.append('filter[store_id]', storeId);
    }
    
    const response = await fetch(url, {
      method: 'GET',
      headers: headers
    });

    if (!response.ok) {
      throw new Error(`HTTP error! Status: ${response.status}`);
    }

    const data = await response.json();
    
    console.log('Page info:', {
      currentPage: data.meta.page.currentPage,
      totalPages: data.meta.page.totalPages,
      totalProducts: data.meta.page.total
    });
    
    // Display product information
    data.data.forEach(product => {
      const attributes = product.attributes;
      console.log(`
        Product ID: ${product.id}
        Name: ${attributes.name}
        Description: ${attributes.description.substring(0, 100)}...
        Status: ${attributes.status}
        Price: $${(attributes.price / 100).toFixed(2)} ${attributes.currency.toUpperCase()}
        Buy Now URL: ${attributes.buy_now_url}
      `);
    });
    
    return {
      products: data.data,
      pagination: {
        currentPage: data.meta.page.currentPage,
        totalPages: data.meta.page.totalPages,
        totalItems: data.meta.page.total
      }
    };
  } catch (error) {
    console.error('Error fetching products:', error);
    throw error;
  }
}

// Usage with store ID
const STORE_ID = 'your_store_id_here';

listProducts(STORE_ID, 1)
  .then(result => {
    console.log('Fetched products:', result.products);
    console.log('Pagination info:', result.pagination);
  })
  .catch(error => {
    console.error('Failed to fetch products:', error);
  });
