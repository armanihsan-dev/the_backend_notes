NGINX: location and autoindex Directives

Part 1: The location Directive

Definition

The location directive is used to define how NGINX handles requests for specific URIs. It allows you to apply different configurations based on the request path.

Syntax

location [modifier] pattern {
    # Configuration directives
}

Modifiers (Matching Rules)

Modifier Description Priority
= Exact match Highest (1)
^~ Preferential prefix match High (2)
~ Case-sensitive regex Medium (3)
~* Case-insensitive regex Medium (3)
(none) Prefix match Lowest (4)

How It Works: Order of Evaluation

# Example showing all types
server {
    # 1. EXACT MATCH (Highest priority)
    location = /exact-path {
        return 200 "Exact match only";
    }
    
    # 2. PREFERENTIAL PREFIX (High priority)
    location ^~ /static/ {
        # Matches /static/, /static/file.html, etc.
        alias /var/www/static/;
    }
    
    # 3. REGEX MATCHES (Medium priority - first match wins)
    location ~ \.php$ {
        # Case-sensitive: matches .php files
        fastcgi_pass php-fpm;
    }
    
    location ~* \.(jpg|png|gif)$ {
        # Case-insensitive: matches image files
        expires 30d;
    }
    
    # 4. PREFIX MATCH (Lowest priority)
    location / {
        # Catches everything else
        root /var/www/html;
        index index.html;
    }
}

Practical Examples

Example 1: API vs Static Files

server {
    listen 80;
    server_name example.com;
    
    # API endpoints (regex match)
    location ~ ^/api/ {
        proxy_pass http://backend:3000;
        proxy_set_header Host $host;
    }
    
    # Static files (preferential prefix)
    location ^~ /static/ {
        root /var/www;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # Default (prefix match)
    location / {
        root /var/www/html;
        try_files $uri $uri/ /index.html;
    }
}

Example 2: Admin Area Protection

location /admin/ {
    # Require authentication
    auth_basic "Admin Area";
    auth_basic_user_file /etc/nginx/.htpasswd;
    
    # Only allow specific IPs
    allow 192.168.1.0/24;
    allow 10.0.0.1;
    deny all;
}

Part 2: The autoindex Directive

Definition

The autoindex directive enables directory listing when the requested URL maps to a directory and no index file is found.

Syntax & Options

location /path/ {
    autoindex on | off;  # Default: off
    
    # Optional enhancements
    autoindex_exact_size on | off;
    autoindex_format html | xml | json | jsonp;
    autoindex_localtime on | off;
}

Key Directives

Directive Values Default Description
autoindex on, off off Enable/disable directory listing
autoindex_exact_size on, off on on: bytes, off: KB/MB/GB
autoindex_format html, xml, json, jsonp html Output format
autoindex_localtime on, off off on: local time, off: UTC

Basic Example

server {
    listen 80;
    server_name files.example.com;
    root /var/www/files;
    
    location /downloads/ {
        autoindex on;
        autoindex_exact_size off;  # Human-readable sizes
        autoindex_localtime on;    # Show local time
        
        # Optional: Add custom headers
        add_header Content-Type "text/html; charset=utf-8";
    }
}

Part 3: Combined Use Cases

Use Case 1: File Sharing Server

server {
    listen 80;
    server_name share.local;
    root /var/www/shared;
    
    # Enable directory listing for all paths
    location / {
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
        
        # Style the autoindex
        autoindex_format html;
        add_before_body /autoindex/header.html;
        add_after_body /autoindex/footer.html;
    }
    
    # Protect sensitive directories
    location /private/ {
        autoindex off;  # Explicitly disable
        deny all;       # Block access completely
    }
    
    # Disable listing for specific directory but allow file access
    location /confidential/ {
        autoindex off;
        # Files are accessible, but directory won't list contents
    }
}

Use Case 2: Development Environment

server {
    listen 8080;
    root /home/dev/projects;
    
    # Enable autoindex for all project directories
    location /projects/ {
        autoindex on;
        autoindex_format html;
        
        # Show file sizes in human-readable format
        autoindex_exact_size off;
        
        # Customize appearance
        location ~ /projects/(.*)/docs/ {
            # Different settings for docs folders
            autoindex on;
            autoindex_format json;  # API-like output for docs
        }
    }
    
    # API endpoints don't need autoindex
    location ~ ^/api/ {
        autoindex off;
        proxy_pass http://localhost:3000;
    }
}

Use Case 3: Media Server with Thumbnails

server {
    listen 80;
    server_name media.local;
    root /var/www/media;
    
    # Photo galleries - show as directory listing
    location /photos/ {
        autoindex on;
        autoindex_format html;
        
        # Add custom CSS for better presentation
        location = /photos/autoindex.css {
            alias /etc/nginx/autoindex-theme.css;
        }
    }
    
    # Videos directory - no listing (streaming only)
    location /videos/ {
        autoindex off;
        
        # MP4 streaming
        location ~ \.mp4$ {
            mp4;
            mp4_buffer_size 1m;
            mp4_max_buffer_size 5m;
        }
    }
}

Part 4: Common Patterns & Best Practices

Pattern 1: Selective Autoindex

# Enable autoindex only for specific directories
location ~ ^/(downloads|public|shared)/ {
    autoindex on;
    autoindex_exact_size off;
    
    # Don't list hidden files
    location ~ /\. {
        deny all;
    }
}

# Everything else - no autoindex
location / {
    autoindex off;
    try_files $uri $uri/ =404;
}

Pattern 2: JSON API for Directory Listing

location /api/directory/ {
    autoindex on;
    autoindex_format json;
    
    # Add CORS headers for web apps
    add_header 'Access-Control-Allow-Origin' '*';
    add_header 'Content-Type' 'application/json';
    
    # Remove HTML wrapper
    location = /api/directory/ {
        rewrite ^ /api/directory/index.json last;
    }
}

Pattern 3: Styling Autoindex Pages

location /files/ {
    autoindex on;
    
    # Inject custom HTML/CSS
    add_before_body /theme/header.html;
    add_after_body /theme/footer.html;
    
    # Custom location for CSS
    location = /files/autoindex.css {
        alias /etc/nginx/custom-autoindex.css;
    }
    
    # Custom location for header/footer
    location ^~ /theme/ {
        internal;  # Can't be accessed directly
        alias /etc/nginx/autoindex-theme/;
    }
}

Common Issues & Solutions

Issue 1: Autoindex not working

# WRONG - root at wrong level
location /downloads {
    root /var/www;  # Looks for /var/www/downloads
    autoindex on;
}

# RIGHT
location /downloads {
    alias /var/www/downloads;  # Maps /downloads to this path
    autoindex on;
}

Issue 2: Conflict with index directive

location / {
    root /var/www/html;
    index index.html;  # This prevents autoindex if index.html exists
    autoindex on;      # Only works if index.html is NOT found
}

Issue 3: Multiple location blocks interfering

# This location might catch everything first
location ~ \.html$ {
    # Processes HTML files
}

# This location for directories
location / {
    autoindex on;  # Might not work for HTML files
}

Quick Reference Cheatsheet

Location Matching Priority

  • 1. = (exact match)
  • 2. ^~ (preferential prefix)
  • 3. ~ or ~* (regex - first match wins)
  • 4. No modifier (prefix match)

Autoindex Best Practices

  • Use alias not root for specific location blocks
  • Set autoindex_exact_size off for user-friendly displays
  • Use autoindex_localtime on for local time zones
  • Consider security: disable autoindex for sensitive directories
  • Add authentication for public-facing directories

Security Considerations

# NEVER do this in production without protection
location / {
    autoindex on;
    # Anyone can browse your entire filesystem!
}

# INSTEAD do this
location /public/ {
    autoindex on;
    # Only public files are exposed
    auth_basic "Restricted";
    auth_basic_user_file /etc/nginx/.htpasswd;
}
