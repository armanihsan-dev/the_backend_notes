Redis vs MongoDB: Node.js Speed Test

Comparing the speed of **Redis** and **MongoDB** using Node.js involves performing identical data operations on both systems and accurately measuring the time taken for each. This comparison typically focuses on basic **CRUD (Create, Read, Update, Delete)** operations.

Since **Redis is an in-memory data store** and **MongoDB is primarily disk-based** (though it uses a significant in-memory cache), Redis is generally expected to be **significantly faster** for simple read/write operations, especially when the working dataset fits in memory.

Here is a conceptual and clear code structure using Node.js to perform a simple speed test for inserting and reading data.

🛠️ Prerequisites

Before running the test, ensure you have:

  • **Node.js** installed.
  • Running instances of **Redis** and **MongoDB**.
  • The necessary npm packages installed in your project:
    • `mongoose` (for MongoDB)
    • `redis` (or a modern equivalent like `@redis/client`)
    • `faker-js` (or similar for generating test data)
Bash
npm init -y
npm install mongoose redis @faker-js/faker

⚙️ Project Setup: speed-test.js

This script will perform a batch of operations and measure the elapsed time. We will use a simple object for comparison.

1. Imports and Configuration

JavaScript
// speed-test.js
const mongoose = require('mongoose');
const redis = require('redis');
const { faker } = require('@faker-js/faker');

// --- Configuration ---
const NUM_OPERATIONS = 10000;
const MONGO_URI = 'mongodb://localhost:27017/speedTestDB';

// Simple data object schema for testing
const TestSchema = new mongoose.Schema({
    key: { type: String, required: true, unique: true },
    value: { type: String, required: true },
    timestamp: { type: Date, default: Date.now }
});
const TestModel = mongoose.model('Test', TestSchema);

2. Connection Functions

JavaScript
// --- Connection Setup ---

// MongoDB Connection
async function connectMongoDB() {
    await mongoose.connect(MONGO_URI);
    console.log('MongoDB connected.');
    // Clear collection for a clean test run
    await TestModel.deleteMany({}); 
}

// Redis Connection
const redisClient = redis.createClient();
redisClient.on('error', (err) => console.log('Redis Client Error', err));

async function connectRedis() {
    await redisClient.connect();
    console.log('Redis connected.');
    // Clear all keys for a clean test run
    await redisClient.flushAll(); 
}

3. Test Helper Function

This function handles the time measurement for a specific database operation.

JavaScript
// --- Helper Function for Timing ---
async function runTest(dbName, operationName, operationFn) {
    const startTime = process.hrtime.bigint();
    
    try {
        await operationFn();
        
        const endTime = process.hrtime.bigint();
        const elapsedTime = Number(endTime - startTime) / 1000000; // Convert nanoseconds to milliseconds
        
        console.log(`\n✅ ${dbName} - ${operationName}:`);
        console.log(`   - Time taken for ${NUM_OPERATIONS} operations: ${elapsedTime.toFixed(2)} ms`);
        console.log(`   - Operations per second: ${(NUM_OPERATIONS / (elapsedTime / 1000)).toFixed(2)} ops/sec`);
        
    } catch (error) {
        console.error(`\n❌ ${dbName} - ${operationName} FAILED:`, error.message);
    }
}

4. Operation Functions

We'll define functions for **inserting** and **reading** a set of data. The keys will be stored to ensure we read the exact same data we inserted.

JavaScript
// --- Data Preparation ---
let testDataKeys = [];
const testData = Array.from({ length: NUM_OPERATIONS }, (_, i) => {
    const key = `user:${i}`;
    testDataKeys.push(key);
    return {
        key: key,
        value: faker.lorem.sentences(3),
        timestamp: new Date()
    };
});

// --- MongoDB Operations ---
async function mongoInsertTest() {
    // Insert documents one by one to simulate individual operations
    const insertPromises = testData.map(data => TestModel.create(data));
    await Promise.all(insertPromises);
}

async function mongoReadTest() {
    // Read documents one by one using the indexed 'key'
    const readPromises = testDataKeys.map(key => TestModel.findOne({ key: key }).exec());
    await Promise.all(readPromises);
}

// --- Redis Operations (using HSET for structured data simulation) ---
async function redisInsertTest() {
    const insertPromises = testData.map(data => {
        // Use HSET to store the structured data as a Redis Hash
        return redisClient.hSet(data.key, {
            key: data.key,
            value: data.value,
            timestamp: data.timestamp.toISOString() // Convert Date to string for Redis
        });
    });
    await Promise.all(insertPromises);
}

async function redisReadTest() {
    // Read documents one by one using the keys
    const readPromises = testDataKeys.map(key => redisClient.hGetAll(key));
    await Promise.all(readPromises);
}

5. Main Execution Block

JavaScript
// --- Main Execution ---
async function main() {
    await connectMongoDB();
    await connectRedis();

    console.log(`\n--- Starting Speed Test with ${NUM_OPERATIONS} operations ---`);
    console.log(`Data size per record is: ~${Buffer.byteLength(JSON.stringify(testData[0]), 'utf8')} bytes.`);
    
    // 1. Write Tests (Insert)
    await runTest('MongoDB', 'Sequential Inserts', mongoInsertTest);
    await runTest('Redis', 'Sequential Inserts', redisInsertTest);

    // 2. Read Tests (Fetch)
    await runTest('MongoDB', 'Sequential Reads', mongoReadTest);
    await runTest('Redis', 'Sequential Reads', redisReadTest);

    // Cleanup
    await mongoose.disconnect();
    await redisClient.quit();
    console.log('\n--- Test Complete and Connections Closed ---');
}

main().catch(console.error);

📈 Expected Results and Analysis

When you run this code (`node speed-test.js`), you will likely see a significant difference in performance.

Operation Expected Winner Reason
**Insert** **Redis** In-memory operations are much faster than disk-based writes, even with MongoDB's journaling.
**Read** **Redis** Accessing data directly from RAM is inherently faster than retrieving data from MongoDB's memory cache (which still has more overhead for document parsing and structure).

**Redis** typically operates at sub-millisecond latency because it:

  • Is **in-memory-only** (data is in RAM).
  • Is **single-threaded** and uses an efficient event loop model.
  • Performs **simple key-value (or key-hash) lookups**.

**MongoDB** offers richer functionality, like flexible document schema and complex querying, which comes with more overhead. However, it's designed for **persistence** and **scaling large datasets** that don't fit in memory.

The key takeaway is that for **simple, high-speed, high-throughput key-based operations** (like caching or session management), **Redis will almost always be faster**. For persistent storage of **complex documents** and **rich query capabilities**, MongoDB is the better choice.
